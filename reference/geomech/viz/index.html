    viz ¶

    3-D visualizer and HTML report for systems derived with geomech.

    Scene declares rigid bodies, point masses, rods and traces with geomech expressions of the configuration variables and parameters, evaluated through geomech.codegen.lambdify on the system's StateLayout. Report puts the animation of a Trajectory, the Lagrangian, the equations of motion, the standard form, the verification numbers and the parameters into one self-contained HTML page (three.js and MathJax from pinned CDN URLs). latex_names gives readable LaTeX for the leaf names (J1 → J_1). See docs/visualization.md.

    import geomech does not import this package, and this package does not import scipy.

    Box(lx, ly, lz) dataclass ¶

    A box with edges lx, ly, lz along the body axes e₁, e₂, e₃.

    Give it a non-square cross-section (lx != ly for a link along e₃) so that a spin about the link's own axis is visible.

    Cylinder(radius, length, axis='z') dataclass ¶

    A cylinder of radius and length along the body axis "x", "y" or "z".

    PointMass(position, radius=None, color=None, name=None) dataclass ¶

    A point mass drawn as a sphere at position (a vector expression).

    radius=None (the default) sizes the sphere from the scene when the report is built: 6 % of the scene's half-size, at most 35 % of the largest body.

    def __init__(
    self,
    title: str,
    system: tuple,
    eom: dict | None = None,
    standard_form: dict | None = None,
    scene: Scene | None = None,
    trajectory: Trajectory | None = None,
    params: dict | None = None,
    checks: bool = True,
    *,
    units: Mapping[str, str] | None = None,
    source: str | None = None,
    derive_time: float | None = None,
    fps: float = 60.0,
    description: str = "",
):
    from geomech import compute_eom, to_standard_form

    if not isinstance(system, (tuple, list)) or len(system) != 4:
        raise TypeError("Report: system must be (L, dW, variables, inputs)")
    if not (fps > 0 and np.isfinite(fps)):
        raise ValueError(f"Report: fps must be positive, got {fps}")
    if (scene is None) != (trajectory is None):
        raise ValueError("Report: pass both scene and trajectory, or neither")
    self.title = str(title)
    self.description = str(description)
    self.L, self.dW, self.variables, inputs = system
    self.inputs = list(inputs or [])
    if eom is None:
        t0 = time.perf_counter()
        eom = compute_eom(self.L, self.dW, self.variables)
        derive_time = time.perf_counter() - t0
    self.eom = eom
    self.derive_time = derive_time
    if standard_form is None:
        standard_form = to_standard_form(eom, self.variables, self.inputs)
    self.standard_form = standard_form
    self.scene: Any = scene
    self.trajectory: Any = trajectory
    if trajectory is not None:
        from geomech.viz.scene import check_finite_states, check_same_params

        simulated = dict(trajectory.p)
        check_same_params(params, simulated, "Report(params=...)")
        if scene is not None:
            check_same_params(scene.params, simulated, "Scene(params=...)")
            check_finite_states(
                np.asarray(trajectory.state, dtype=float),
                np.asarray(trajectory.t, dtype=float),
                scene.layout,
                "Report",
            )
        params = simulated
    self.params = dict(params or {})
    self.units = dict(units or {})
    self.source = source
    self.fps = float(fps)
    self.check_error: float | None = None
    self.check_time: float | None = None
    if checks:
        from geomech.numeric import check_eom

        t0 = time.perf_counter()
        self.check_error = float(
            check_eom(self.L, self.dW, self.variables, self.eom, inputs=self.inputs)
        )
        self.check_time = time.perf_counter() - t0
    self.names = latex_names(
        self.L,
        self.dW,
        self.eom,
        self.standard_form,
        list(self.variables.scalars),
        list(self.variables.vectors),
        list(self.variables.matrices),
        self.inputs,
    )
    self._html: str | None = None

    def energy_drift(self) -> float | None:
    """``max |E(t) − E(0)| / |E(0)|`` of ``trajectory.energy()`` (``None`` without an
    energy function); the absolute drift when ``E(0) = 0``."""
    try:
        E = np.asarray(self.trajectory.energy(), dtype=float)
    except TypeError:
        return None
    scale = abs(E[0]) if abs(E[0]) > 0 else 1.0
    return float(np.max(np.abs(E - E[0])) / scale)

    def forced(self) -> bool:
    """``True`` when the trajectory has a nonzero input at some stored state
    (``trajectory.u``); the energy then changes by the work of the inputs. ``False``
    both for zero inputs and for inputs that were not recorded (``u is None``, e.g. a
    hand-built ``Trajectory``); the summary calls only the first "unforced"."""
    u = getattr(self.trajectory, "u", None)
    return u is not None and bool(np.any(np.asarray(u) != 0))

    def frame_indices(self) -> np.ndarray:
    """The stored states kept in the animation: the first state at or after each
    multiple of ``1/fps`` of simulated time, and the last state."""
    t = np.asarray(self.trajectory.t, dtype=float)
    if len(t) < 2:
        return np.arange(len(t))
    targets = np.arange(t[0], t[-1], 1.0 / self.fps)
    idx = np.searchsorted(t, targets - 1e-12)
    return np.unique(np.r_[idx[idx < len(t)], len(t) - 1])

    def scene_data(self) -> dict:
    """The animation data inlined in the page.

    Positions keep six significant figures of the scene size, rotation entries five
    decimals. The view is framed on everything drawn (shapes, body-axis arrows, point
    and rod radii), with no minimum size, so a 1 mm and a 1 km scene look alike.
    Default point and rod radii are fractions of the scene size.
    """
    scene = self.scene
    idx = self.frame_indices()
    fr = scene.frames(self.trajectory, indices=idx)
    lo, hi = fr.bounds()
    body_ext = max((b.shape.extent for b in scene.bodies), default=0.0)
    explicit = [body_ext / 2]
    explicit += [pm.radius for pm in scene.points if pm.radius is not None]
    explicit += [r.radius for r in scene.rods if r.radius is not None]
    size = max(float(np.max(hi - lo)) / 2, *explicit)
    if not size > 0:
        size = 1.0
    point_r, rod_r = 0.06 * size, 0.01 * size
    if body_ext > 0:
        point_r, rod_r = min(point_r, 0.35 * body_ext), min(rod_r, 0.06 * body_ext)
    points_r = [pm.radius if pm.radius is not None else point_r for pm in scene.points]
    rods_r = [r.radius if r.radius is not None else rod_r for r in scene.rods]

    # bounds of everything drawn: positions widened by each element's reach
    los, his = [lo[None, :]], [hi[None, :]]

    def widen(xyz, reach):
        xyz = np.asarray(xyz, dtype=float).reshape(-1, 3)
        los.append(xyz.min(axis=0, keepdims=True) - reach)
        his.append(xyz.max(axis=0, keepdims=True) + reach)

    reach_of = []
    for body, ev in zip(scene.bodies, fr.bodies):
        lengths = _axis_lengths(body.shape)
        reach = float(np.linalg.norm(_half_sizes(body.shape)))
        if body.axes:
            reach = max(reach, max(lengths))
        reach_of.append(lengths)
        widen(ev["center"], reach)
    for v, r in zip(fr.points, points_r):
        widen(v, r)
    for v, r in zip(fr.rods, rods_r):
        widen(v, r)
    lo = np.min(np.concatenate(los), axis=0)
    hi = np.max(np.concatenate(his), axis=0)
    center = (lo + hi) / 2
    radius = float(np.max(hi - lo)) / 2
    if not radius > 0:
        radius = size
    dec = _position_decimals(radius)
    center = np.round(center, dec)
    zlow = float(lo[2])
    ground = 0.0 if 0.0 <= zlow <= 2 * radius else zlow - 0.05 * radius

    rot_tracks: list[list] = []
    rot_keys: dict[str, int] = {}
    bodies = []
    for body, ev, lengths in zip(scene.bodies, fr.bodies, reach_of):
        key = str(body.R)
        if key not in rot_keys:
            rot_keys[key] = len(rot_tracks)
            rot_tracks.append(_flat(ev["rotation"]))
        bodies.append(
            {
                "name": body.name,
                "shape": _shape_dict(body.shape),
                "color": body.color,
                "axes": bool(body.axes),
                "axisLengths": [_sig(v) for v in lengths],
                "rot": rot_keys[key],
                "pos": _flat(ev["center"], dec),
            }
        )
    return {
        "t": _flat(fr.t - fr.t[0]),
        "t0": float(fr.t[0]),
        "rotations": rot_tracks,
        "bodies": bodies,
        "points": [
            {"name": pm.name, "radius": _sig(r), "color": pm.color, "pos": _flat(v, dec)}
            for pm, r, v in zip(scene.points, points_r, fr.points)
        ],
        "rods": [
            {"name": rd.name, "radius": _sig(r), "color": rd.color, "ends": _flat(v, dec)}
            for rd, r, v in zip(scene.rods, rods_r, fr.rods)
        ],
        "traces": [
            {"name": tr.name, "color": tr.color, "pos": _flat(v, dec)}
            for tr, v in zip(scene.traces, fr.traces)
        ],
        "center": [float(c) for c in center],
        "radius": _sig(radius),
        "ground": _sig(ground),
    }

    def stats(self) -> list[_Stat]:
    """The numbers of the summary strip."""
    out = []
    if self.derive_time is not None:
        out.append(_Stat("Derivation", f"{self.derive_time:.3g} s", note="compute_eom"))
    else:
        out.append(_Stat("Derivation", "—", note="compute_eom (not timed)"))
    if self.check_error is None:
        out.append(_Stat("check_eom", "not run", "bad"))
    else:
        ok = self.check_error <= CHECK_EOM_TOL
        out.append(
            _Stat(
                "check_eom error",
                f"{self.check_error:.2e}",
                "ok" if ok else "bad",
                note=f"{'passes' if ok else 'FAILS'} (tolerance {CHECK_EOM_TOL:g})",
            )
        )
    traj = self.trajectory
    if traj is not None:
        drift = self.energy_drift()
        forced = self.forced()
        # "drift" (an integrator error) only for a run known to be unforced: zero
        # recorded inputs. Unknown inputs (traj.u is None) give a neutral "change".
        recorded = getattr(traj, "u", None) is not None
        label = "Energy drift" if recorded and not forced else "Energy change"
        if drift is None:
            out.append(_Stat(label, "—", note="module has no energy()"))
        elif forced:
            out.append(
                _Stat(
                    label,
                    f"{drift:.2e}",
                    note="max |E − E₀| / |E₀| of a forced run: the work of the inputs,"
                    " not an integrator error",
                )
            )
        elif not recorded:
            out.append(
                _Stat(label, f"{drift:.2e}", note="max |E − E₀| / |E₀|; inputs not recorded")
            )
        else:
            out.append(_Stat(label, f"{drift:.2e}", note="max |E − E₀| / |E₀|, unforced"))
        out.append(
            _Stat("Constraint drift", f"{traj.max_drift():.1e}", note="|q|−1, ‖RᵀR − I‖")
        )
        dt = traj.info.get("dt")
        step = f"dt = {dt:.3g} s" if dt is not None else ""
        out.append(_Stat("Integrator", traj.method, note=step))
        t = np.asarray(traj.t)
        out.append(_Stat("Simulated", f"{t[-1] - t[0]:.3g} s", note=f"{len(t)} stored states"))
    return out

    def add(self, element):
    """Add a ``RigidBody``, ``PointMass``, ``Rod`` or ``Trace``; returns the element."""
    from geomech.core.base.types import ExprType

    if isinstance(element, RigidBody):
        owner = f"RigidBody '{element.name}'"
        R = _as_expr(element.R, owner, "R")
        if R.type != ExprType.MATRIX:
            raise ValueError(f"{owner}: R '{R}' must be a matrix (an SO3 variable)")
        _leaf_kinds(R, self.layout, owner, "attitude R")
        rot = _Evaluated(R, self.layout, "matrix", owner)
        origin = self._vector(element.origin, owner, "origin")
        offset = self._offset(element.offset, owner)
        if element.color is None:
            element.color = _DEFAULT_BODY[len(self.bodies) % len(_DEFAULT_BODY)]
        self._compiled[id(element)] = (rot, origin, offset)
        self.bodies.append(element)
    elif isinstance(element, PointMass):
        owner = f"PointMass '{element.name}'"
        self._compiled[id(element)] = (self._vector(element.position, owner, "position"),)
        if element.color is None:
            element.color = "load"
        self.points.append(element)
    elif isinstance(element, Rod):
        owner = f"Rod '{element.name}'"
        a = self._vector(element.a, owner, "end point a")
        b = self._vector(element.b, owner, "end point b")
        self._compiled[id(element)] = (a, b)
        if element.color is None:
            element.color = "cable"
        self.rods.append(element)
    elif isinstance(element, Trace):
        owner = f"Trace '{element.name}'"
        self._compiled[id(element)] = (self._vector(element.point, owner, "point"),)
        if element.color is None:
            element.color = "trace"
        self.traces.append(element)
    else:
        raise TypeError(
            "Scene.add: expected a RigidBody, PointMass, Rod or Trace, got"
            f" {type(element).__name__}"
        )
    return element

    def frames(self, traj, indices: Sequence[int] | np.ndarray | None = None) -> SceneFrames:
    """The scene evaluated at every stored state of *traj* (a ``Trajectory`` or an
    ``(N, n_state)`` array), or at the rows *indices* of it.

    A ``Trajectory`` is evaluated with its own parameters ``traj.p``; a bare array
    with ``Scene(params=...)``.

    Raises ``ValueError`` when a state is not finite (naming the first time and state
    entry), when ``Scene(params=...)`` contradicts ``traj.p``, or when an attitude is
    not a rotation (``RᵀR ≠ I`` or ``det R ≠ 1`` beyond 1e-6) at some state.
    """
    t, states = self._states(traj)
    rows = None
    if indices is not None:
        rows = np.asarray(indices, dtype=int)
        t, states = t[rows], states[rows]
    check_finite_states(states, t, self.layout, "Scene.frames", rows)
    p = self.params_for(traj)
    out = SceneFrames(t=t)
    for body in self.bodies:
        rot, origin, offset_fn = self._compiled[id(body)]
        R = rot(states, p)
        err = np.abs(np.einsum("nki,nkj->nij", R, R) - np.eye(3)).max(initial=0.0)
        det = np.linalg.det(R) if len(R) else np.ones(0)
        if err > _ORTHO_TOL or np.any(np.abs(det - 1.0) > _ORTHO_TOL):
            raise ValueError(
                f"Scene.frames: the attitude '{body.R}' of '{body.name}' is not a rotation"
                f" (max |RᵀR − I| = {err:.2e}, min det = {det.min():.6f})"
            )
        pos = origin(states, p)
        offset = offset_fn(p)
        center = pos + np.einsum("nij,j->ni", R, offset)
        out.bodies.append({"position": pos, "rotation": R, "center": center})
    for pm in self.points:
        (pos,) = self._compiled[id(pm)]
        out.points.append(pos(states, p))
    for rod in self.rods:
        a, b = self._compiled[id(rod)]
        out.rods.append(np.stack([a(states, p), b(states, p)], axis=1))
    for tr in self.traces:
        (pt,) = self._compiled[id(tr)]
        out.traces.append(pt(states, p))
    return out

    def bounds(self) -> tuple[np.ndarray, np.ndarray]:
    """The smallest and largest inertial coordinates of every evaluated point."""
    pts = [b["position"] for b in self.bodies] + [b["center"] for b in self.bodies]
    pts += list(self.points) + [r.reshape(-1, 3) for r in self.rods] + list(self.traces)
    if not pts:
        return np.zeros(3), np.zeros(3)
    allp = np.concatenate([np.asarray(p).reshape(-1, 3) for p in pts])
    return allp.min(axis=0), allp.max(axis=0)

    def latex_names(*objs: Iterable) -> dict[str, str]:
    r"""``{leaf name: LaTeX}`` for every leaf of *objs* whose printed form changes.

    *objs* are expressions, the dicts of ``compute_eom`` or ``to_standard_form``, the
    ``SystemVariables`` lists or any list of these. The tangent and variation vectors of
    each S² and SO(3) point in *objs* are included. A name is left out when its pretty
    form would collide with another leaf's name or pretty form, so the map is one-to-one
    on the leaves it sees. Pass the result to ``to_latex(expr, names=...)``.
    """
    from geomech.core.base.expressions import S2, SO3

    leaves: dict = {}
    for obj in objs:
        _gather(obj, leaves)
    # the velocity and variation vectors of every manifold point, whether or not they occur
    for leaf in list(leaves.values()):
        if isinstance(leaf, (S2, SO3)):
            for vec in (leaf.get_tangent_vector(), leaf.get_variation_vector()):
                leaves.setdefault(str(vec), vec)
    proposed = {name: pretty_name(name, _point_of(leaf)) for name, leaf in leaves.items()}
    # one-to-one: a target shared by two leaves, or equal to another leaf's own name, is
    # dropped for every leaf involved
    count: dict[str, int] = {}
    for target in proposed.values():
        count[target] = count.get(target, 0) + 1
    out = {}
    for name, target in proposed.items():
        if target == name:
            continue
        if count[target] > 1 or (target in leaves and target != name):
            continue
        out[name] = target
    return out

    def pretty_name(name: str, point: str | None = None) -> str:
    r"""The LaTeX of one leaf name (see the module docstring for the rules).

    *point* is the name of the S² or SO(3) point when the leaf is its tangent or variation
    vector (``\Omega_{R1}`` of ``R1``); the vector then takes the point's index
    (``\Omega_1``). Without *point* such a name is left as it is.
    """
    if point is not None:
        m = _TANGENT.match(name)
        p = _indexed(point)
        if m and m.group(2) == point and p:
            return _subscript(m.group(1), p[1])
        return name
    idx = _indexed(name)
    if idx:
        return _subscript(*idx)
    return name
