    scene ¶

    The 3-D scene of a system: rigid bodies, point masses, rods and traces.

    Geometry is declared with geomech expressions of the configuration variables and the parameters, and evaluated with geomech.codegen.lambdify on the system's StateLayout: the visualizer has no kinematics of its own.

    .. code-block:: python

    scene = Scene(layout)            # parameters come from the trajectory
scene.add(RigidBody(R1, origin=ZeroVector, shape=Box(0.06, 0.03, 1.0), offset=[0, 0, -0.5]))
scene.add(PointMass(x_L, radius=0.05))
scene.add(Rod(x_Q, x_L))
scene.add(Trace(R1 * l1 + R2 * l2))
frames = scene.frames(traj)      # positions and rotations per stored state, with traj.p

    Box(lx, ly, lz) dataclass ¶

    A box with edges lx, ly, lz along the body axes e₁, e₂, e₃.

    Give it a non-square cross-section (lx != ly for a link along e₃) so that a spin about the link's own axis is visible.

    Cylinder(radius, length, axis='z') dataclass ¶

    A cylinder of radius and length along the body axis "x", "y" or "z".

    PointMass(position, radius=None, color=None, name=None) dataclass ¶

    A point mass drawn as a sphere at position (a vector expression).

    radius=None (the default) sizes the sphere from the scene when the report is built: 6 % of the scene's half-size, at most 35 % of the largest body.

    RigidBody(R, origin, shape, offset=None, axes=True, color=None, name=None) dataclass ¶

    A rigid body drawn as a solid shape with its body frame.

    Parameters:

    Name Type Description Default
    R Any

    the attitude, an SO3 configuration variable (or a matrix expression of the configuration variables and parameters that evaluates to a rotation).

    		 required
    origin Any

    the body's reference point in the inertial frame, a vector expression (ZeroVector for a body hinged at the origin).

    		 required
    shape Any

    a Box, Cylinder or Sphere, rotated by R.

    		 required
    offset Any

    where the shape's centre sits in the body frame: the shape is centred at origin + R offset. A 3-sequence of numbers or a constant vector expression (a parameter such as \rho); None is zero.

    		 None
    axes bool

    draw the body frame (e₁ red, e₂ green, e₃ blue) at the shape's centre.

    		 True
    color str | None

    a palette name ("link1" … "link4", "load", "rotor", "cable", "trace") or "#rrggbb"; None picks the next link colour.

    		 None
    name str | None

    a label (defaults to str(R)).

    		 None

    Rod(a, b, radius=None, color=None, name=None) dataclass ¶

    A rod (cable, massless link) between two vector expressions a and b.

    radius=None (the default) sizes the rod from the scene when the report is built: 1 % of the scene's half-size, at most 6 % of the largest body.

    Scene(layout, params=None) ¶

    A set of bodies, point masses, rods and traces drawn from a trajectory.

    Parameters:

    Name Type Description Default
    layout StateLayout

    the StateLayout the simulated module was generated from.

    		 required
    params dict | None

    optional. frames(traj) of a Trajectory always evaluates the scene with traj.p, the parameters that were simulated, and raises ValueError when params gives a different value for one of them. Only needed to evaluate bare state arrays (frames(states)).

    		 None

    Every expression may use configuration variables and parameters of layout only; anything else raises ValueError naming the leaf when the element is added.

    def __init__(self, layout: StateLayout, params: dict | None = None):
    from geomech.codegen import StateLayout

    if not isinstance(layout, StateLayout):

    self.params = dict(params or {})

    self.points: list[PointMass] = []
    self.rods: list[Rod] = []

    def add(self, element):
    """Add a ``RigidBody``, ``PointMass``, ``Rod`` or ``Trace``; returns the element."""
    from geomech.core.base.types import ExprType

    if isinstance(element, RigidBody):
        owner = f"RigidBody '{element.name}'"
        R = _as_expr(element.R, owner, "R")
        if R.type != ExprType.MATRIX:
            raise ValueError(f"{owner}: R '{R}' must be a matrix (an SO3 variable)")
        _leaf_kinds(R, self.layout, owner, "attitude R")
        rot = _Evaluated(R, self.layout, "matrix", owner)
        origin = self._vector(element.origin, owner, "origin")
        offset = self._offset(element.offset, owner)
        if element.color is None:
            element.color = _DEFAULT_BODY[len(self.bodies) % len(_DEFAULT_BODY)]
        self._compiled[id(element)] = (rot, origin, offset)
        self.bodies.append(element)
    elif isinstance(element, PointMass):
        owner = f"PointMass '{element.name}'"
        self._compiled[id(element)] = (self._vector(element.position, owner, "position"),)
        if element.color is None:
            element.color = "load"
        self.points.append(element)
    elif isinstance(element, Rod):
        owner = f"Rod '{element.name}'"
        a = self._vector(element.a, owner, "end point a")
        b = self._vector(element.b, owner, "end point b")
        self._compiled[id(element)] = (a, b)
        if element.color is None:
            element.color = "cable"
        self.rods.append(element)
    elif isinstance(element, Trace):
        owner = f"Trace '{element.name}'"
        self._compiled[id(element)] = (self._vector(element.point, owner, "point"),)
        if element.color is None:
            element.color = "trace"
        self.traces.append(element)
    else:
        raise TypeError(
            "Scene.add: expected a RigidBody, PointMass, Rod or Trace, got"
            f" {type(element).__name__}"
        )
    return element

    def frames(self, traj, indices: Sequence[int] | np.ndarray | None = None) -> SceneFrames:
    """The scene evaluated at every stored state of *traj* (a ``Trajectory`` or an
    ``(N, n_state)`` array), or at the rows *indices* of it.

    A ``Trajectory`` is evaluated with its own parameters ``traj.p``; a bare array
    with ``Scene(params=...)``.

    Raises ``ValueError`` when a state is not finite (naming the first time and state
    entry), when ``Scene(params=...)`` contradicts ``traj.p``, or when an attitude is
    not a rotation (``RᵀR ≠ I`` or ``det R ≠ 1`` beyond 1e-6) at some state.
    """
    t, states = self._states(traj)
    rows = None
    if indices is not None:
        rows = np.asarray(indices, dtype=int)
        t, states = t[rows], states[rows]
    check_finite_states(states, t, self.layout, "Scene.frames", rows)
    p = self.params_for(traj)
    out = SceneFrames(t=t)
    for body in self.bodies:
        rot, origin, offset_fn = self._compiled[id(body)]
        R = rot(states, p)
        err = np.abs(np.einsum("nki,nkj->nij", R, R) - np.eye(3)).max(initial=0.0)
        det = np.linalg.det(R) if len(R) else np.ones(0)
        if err > _ORTHO_TOL or np.any(np.abs(det - 1.0) > _ORTHO_TOL):
            raise ValueError(
                f"Scene.frames: the attitude '{body.R}' of '{body.name}' is not a rotation"
                f" (max |RᵀR − I| = {err:.2e}, min det = {det.min():.6f})"
            )
        pos = origin(states, p)
        offset = offset_fn(p)
        center = pos + np.einsum("nij,j->ni", R, offset)
        out.bodies.append({"position": pos, "rotation": R, "center": center})
    for pm in self.points:
        (pos,) = self._compiled[id(pm)]
        out.points.append(pos(states, p))
    for rod in self.rods:
        a, b = self._compiled[id(rod)]
        out.rods.append(np.stack([a(states, p), b(states, p)], axis=1))
    for tr in self.traces:
        (pt,) = self._compiled[id(tr)]
        out.traces.append(pt(states, p))
    return out

    def params_for(self, traj) -> dict:
    """The parameters the scene is drawn with: ``traj.p`` for a ``Trajectory`` (after
    checking that ``Scene(params=...)`` agrees with it), else the scene's own."""
    simulated = getattr(traj, "p", None)
    if simulated is not None:
        simulated = dict(simulated)
        check_same_params(self.params, simulated, "Scene(params=...)")
        return simulated
    if not self.params and self.layout.param_names and self._needs_params():
        raise ValueError(
            "Scene.frames: the scene uses parameters but has none: pass a Trajectory"
            " (its p is used), or build the Scene with params=p to evaluate bare states"
        )
    return self.params

    def bounds(self) -> tuple[np.ndarray, np.ndarray]:
    """The smallest and largest inertial coordinates of every evaluated point."""
    pts = [b["position"] for b in self.bodies] + [b["center"] for b in self.bodies]
    pts += list(self.points) + [r.reshape(-1, 3) for r in self.rods] + list(self.traces)
    if not pts:
        return np.zeros(3), np.zeros(3)
    allp = np.concatenate([np.asarray(p).reshape(-1, 3) for p in pts])
    return allp.min(axis=0), allp.max(axis=0)

    def check_finite_states(states: np.ndarray, t: np.ndarray, layout, what: str,
                        rows: np.ndarray | None = None) -> None:  # fmt: skip
    """Raise ``ValueError`` naming the first time and state entry that is NaN or
    infinite (a diverged simulation)."""
    finite = np.isfinite(states)
    if finite.all():
        return
    i, j = (int(v) for v in np.argwhere(~finite)[0])
    row = int(rows[i]) if rows is not None else i
    n_bad = int(np.count_nonzero(~finite.all(axis=1)))
    raise ValueError(
        f"{what}: the state at t = {float(t[i]):.6g} (stored state {row}) is not finite:"
        f" {_state_entry(layout, j)} = {states[i, j]}; {n_bad} of {len(states)} states are"
        " non-finite. The simulation diverged: shorten t_span to end before"
        f" t = {float(t[i]):.6g}, reduce dt, or check the model"
    )

    def check_same_params(given: dict | None, simulated: dict, what: str) -> None:
    """Raise ``ValueError`` naming the first parameter whose value in *given* differs from
    the simulated one (``trajectory.p``); parameters missing from either dict are not
    compared."""
    given = given or {}
    for key in sorted(given):
        if key in simulated and not _same_value(given[key], simulated[key]):
            raise ValueError(
                f"{what}: the parameter '{key}' is {_show(given[key])} there but"
                f" {_show(simulated[key])} in trajectory.p, the parameters that were"
                " simulated; the scene and the parameter table use trajectory.p, so drop"
                " the params argument or pass the same p"
            )
