    simulation ¶

    Simulation of generated models on their state manifold, and LQR from linearize.

    accelerations solves the standard form for the accelerations (S² rows on the tangent plane); simulate integrates with a Runge–Kutta–Munthe-Kaas method ("rkmk4") or scipy.integrate.solve_ivp and returns a Trajectory; lqr designs a gain about an equilibrium from a generated linearization. Simulation consumes only the modules of geomech.codegen. scipy (extra geomech[sim]) is imported only by method="solve_ivp" and lqr. See docs/simulation.md.

    def error(self, state) -> np.ndarray:
    """The reduced state error ``δx`` of *state* about ``x_eq`` (see the module
    docstring)."""
    x = np.asarray(state, dtype=float).reshape(-1)
    xe = self.x_eq
    cfg, vel = [], []
    for var in self.model.variables:
        if var.manifold == "S2":
            Bq = self.bases[var.key]
            cfg.append(Bq.T @ np.cross(xe[var.cfg], x[var.cfg]))
            vel.append(Bq.T @ (x[var.vel] - xe[var.vel]))
        elif var.manifold == "SO3":
            Re, R = xe[var.cfg].reshape(3, 3), x[var.cfg].reshape(3, 3)
            cfg.append(log_so3(Re.T @ R))
            vel.append(x[var.vel] - xe[var.vel])
        else:
            cfg.append(x[var.cfg] - xe[var.cfg])
            vel.append(x[var.vel] - xe[var.vel])
    return np.concatenate(cfg + vel)

    def __getitem__(self, key: str) -> np.ndarray:
    """The values of one state slot over time, by printed symbol (``"q"``,
    ``"\\\\omega_{q}"``) or generated name (``"omega_q"``): ``(N,)``, ``(N, 3)`` or
    ``(N, 3, 3)`` for an SO(3) point."""
    lay = self.module.LAYOUT
    for slot in lay["configuration"] + lay["velocity"]:
        if key in (slot["symbol"], slot["name"]):
            vals = self.state[:, slice(*slot["slice"])]
            shape = tuple(slot["shape"])
            return vals[:, 0] if shape == () else vals.reshape((len(self.t), *shape))
    names = [s["name"] for s in lay["configuration"] + lay["velocity"]]
    raise KeyError(f"Trajectory: no state slot {key!r} (slots: {names})")

    def energy(self, fn: Callable | None = None) -> np.ndarray:
    """``fn(state, p)`` at every output state, ``(N,)``.

    *fn* defaults to the module's ``energy`` (``generate_module(...,
    lagrangian=L)``); ``geomech.codegen.energy_function(L, layout)`` gives one for any
    Lagrangian.
    """
    if fn is None:
        fn = getattr(self.module, "energy", None)
        if fn is None:
            raise TypeError(
                "Trajectory.energy: the module has no energy(); generate it with"
                " generate_module(..., lagrangian=L) or pass energy_function(L, layout)"
            )
    # The generated functions take one state at a time; at about 20 µs per state
    # (double rigid pendulum) this loop is a few percent of the simulation that
    # produced the states, so it is not vectorised.
    return np.fromiter((float(fn(x, self.p)) for x in self.state), float, len(self.state))

    accelerations(module, state, u=None, p=None) ¶

    The accelerations a of a generated standard-form module at (state, u).

    Solves mass @ a + bias + input_matrix @ u = 0. Every S² equation is meaningful only on the tangent plane of its point q (docs/conventions.md), and its acceleration ω̇ satisfies q · ω̇ = 0 (from d/dt (q · ω) = 0 with q̇ · ω = (ω × q) · ω = 0). So each S² row block is replaced by P_q · row plus a multiple of q qᵀ on its own ω̇ column, with right-hand side −P_q (bias + G u), P_q = I − q qᵀ; the couplings of that row to the other accelerations are projected too. This works whether the derived S² mass block is singular (Hat(q) Hat(q) m l²) or already full rank (−m l² I, when the engine used q · ω̇ = 0 itself).

    Scaling, so that the result does not depend on the units: the stabiliser is s q qᵀ with s = ‖P_q D‖_F / √2, D the row's own diagonal mass block. Both forms of D project to −m l² P_q and ‖P_q‖_F = √2, so s = m l²: the normal direction gets the same magnitude as the two tangent directions (for an anisotropic tangent inertia, the root mean square of the two). Then every row of the system and of the right-hand side is divided by its largest entry (row equilibration), and the system is solved with np.linalg.solve. The condition number is taken on the equilibrated matrix, so a light link (m₂/m₁ = 1e-9) or a large one (1 t on a 100 km tether, SI units) is solved to round-off and accepted, while a kinetic energy that is degenerate for every parameter value still raises.

    module: Any, state: ArrayLike, u: ArrayLike | None = None, p: dict | None = None

    Solves ``mass @ a + bias + input_matrix @ u = 0``. Every S² equation is meaningful
    only on the tangent plane of its point ``q`` (``docs/conventions.md``), and its
    acceleration ``ω̇`` satisfies ``q · ω̇ = 0`` (from ``d/dt (q · ω) = 0`` with
    ``q̇ · ω = (ω × q) · ω = 0``). So each S² row block is replaced by ``P_q · row`` plus
    a multiple of ``q qᵀ`` on its own ``ω̇`` column, with right-hand side ``−P_q (bias + G u)``,
    ``P_q = I − q qᵀ``; the couplings of that row to the other accelerations are
    projected too. This works whether the derived S² mass block is singular
    (``Hat(q) Hat(q) m l²``) or already full rank (``−m l² I``, when the engine used
    ``q · ω̇ = 0`` itself).

    Scaling, so that the result does not depend on the units: the stabiliser is
    ``s q qᵀ`` with ``s = ‖P_q D‖_F / √2``, ``D`` the row's own diagonal mass block.
    Both forms of ``D`` project to ``−m l² P_q`` and ``‖P_q‖_F = √2``, so ``s = m l²``:
    the normal direction gets the same magnitude as the two tangent directions (for an
    anisotropic tangent inertia, the root mean square of the two). Then every row of the
    system and of the right-hand side is divided by its largest entry (row
    equilibration), and the system is solved with ``np.linalg.solve``. The condition
    number is taken on the equilibrated matrix, so a light link (``m₂/m₁ = 1e-9``) or a
    large one (1 t on a 100 km tether, SI units) is solved to round-off and accepted,
    while a kinetic energy that is degenerate for every parameter value still raises.

    Args:
        module: a module from ``generate_module`` of a ``to_standard_form`` result.
        state: the flat state (``LAYOUT``: configuration slots, then velocity slots).
        u: the inputs (``None`` for zeros).
        p: the parameters, keyed by printed symbol.

    Returns:
        The flat accelerations, ordered like ``LAYOUT["acceleration"]``.

    Raises:
        np.linalg.LinAlgError: the system is singular, i.e. its condition number after
            row equilibration exceeds ``MAX_COND`` (1e12); the message gives the

    def check_state(
    module: Any, state: ArrayLike, tol: float = STATE_TOL, what: str = "x0"
) -> np.ndarray:
    """Check that *state* lies on the state manifold of *module* and return it flat.

    Each S² point has unit norm and its velocity ``ω`` is orthogonal to it; each SO(3)
    point is orthogonal with determinant +1; every entry is finite. The tolerance is
    *tol* (1e-9). Raises ``ValueError`` naming the slot otherwise.
    """
    model = model_of(module)
    x = model.as_state(state, what)
    if not np.all(np.isfinite(x)):
        raise ValueError(f"{what}: non-finite entries {np.flatnonzero(~np.isfinite(x))}")
    for var in model.s2:
        q, w = x[var.cfg], x[var.vel]
        if abs(np.linalg.norm(q) - 1.0) > tol:
            raise ValueError(
                f"{what}: the S2 slot '{var.key}' must have unit norm, got"
                f" |{var.key}| = {np.linalg.norm(q):.12g}"
            )
        if abs(q @ w) > tol:
            raise ValueError(
                f"{what}: the velocity slot '{var.vel_key}' must be orthogonal to"
                f" '{var.key}', got {var.key}·{var.vel_key} = {q @ w:.3g}"
            )
    for var in model.so3:
        R = x[var.cfg].reshape(3, 3)
        err = np.linalg.norm(R.T @ R - np.eye(3))
        if err > tol:
            raise ValueError(
                f"{what}: the SO3 slot '{var.key}' must be orthogonal, got"
                f" |RᵀR − I| = {err:.3g} (row-major, 9 entries)"
            )
        det = np.linalg.det(R)
        if abs(det - 1.0) > tol:
            raise ValueError(
                f"{what}: the SO3 slot '{var.key}' must have determinant +1, got {det:.12g}"
            )
    return x

    def constraint_drift(model: Model, states: np.ndarray) -> dict:
    """``{name: (N,) array}`` of the manifold constraint violations of *states*."""
    states = np.atleast_2d(states)
    out = {}
    for var in model.s2:
        q, w = states[:, var.cfg], states[:, var.vel]
        name = _name(model, var.key)
        out[f"|{name}|-1"] = np.abs(np.linalg.norm(q, axis=1) - 1.0)
        out[f"{name}.{_name(model, var.vel_key)}"] = np.abs(np.einsum("ij,ij->i", q, w))
    for var in model.so3:
        R = states[:, var.cfg].reshape(-1, 3, 3)
        name = _name(model, var.key)
        RtR = np.einsum("nki,nkj->nij", R, R)
        out[f"{name}^T {name}-I"] = np.linalg.norm(RtR - np.eye(3), axis=(1, 2))
    return out

    lqr(lin_module, x_eq, u_eq, p, Q, R, *, a_eq=None, tol=STATE_TOL, module=None, eq_tol=1e-08) ¶

    The continuous-time LQR gain of a generated linearization about an equilibrium.

    Evaluates M_lin, A_lin, B_lin of lin_module (generate_module of a LinearizedSystem) at (x_eq, a_eq, u_eq), reduces them to the tangent coordinates δx = Tᵀ δx_full (Mr = Tᵀ M T, Ar = Tᵀ A T, Br = Tᵀ B), forms Ā = Mr⁻¹ Ar, B̄ = Mr⁻¹ Br, solves scipy.linalg.solve_continuous_are and returns K = R⁻¹ B̄ᵀ P in an LQRController.

    The S² reduction: the 3-D perturbations ξ and δω of an S² point satisfy ξ · q = δω · q = 0 at an equilibrium (where ω = 0), so each has 2 degrees of freedom. In 3-D the normal direction is an uncontrollable mode with a zero eigenvalue, so the Riccati equation has no stabilising solution. T maps the 2-D tangent coordinates in the basis tangent_basis(q_eq) to the 3-D ones, and the S² rows (whose equations hold only on the tangent plane) are projected with Tᵀ. A singular S² mass block (Hat(q) Hat(q) m l²) becomes invertible in this basis.

    def lqr(
    lin_module: Any,
    x_eq: ArrayLike,
    u_eq: ArrayLike,
    p: dict,
    Q: ArrayLike,
    R: ArrayLike,
    *,
    a_eq: ArrayLike | None = None,
    tol: float = STATE_TOL,
    module: Any = None,
    eq_tol: float = 1e-8,
) -> LQRController:
    """The continuous-time LQR gain of a generated linearization about an equilibrium.

    Evaluates ``M_lin``, ``A_lin``, ``B_lin`` of *lin_module* (``generate_module`` of a
    ``LinearizedSystem``) at ``(x_eq, a_eq, u_eq)``, reduces them to the tangent
    coordinates ``δx = Tᵀ δx_full`` (``Mr = Tᵀ M T``, ``Ar = Tᵀ A T``, ``Br = Tᵀ B``),
    forms ``Ā = Mr⁻¹ Ar``, ``B̄ = Mr⁻¹ Br``, solves ``scipy.linalg.solve_continuous_are``
    and returns ``K = R⁻¹ B̄ᵀ P`` in an ``LQRController``.

    The S² reduction: the 3-D perturbations ``ξ`` and ``δω`` of an S² point satisfy
    ``ξ · q = δω · q = 0`` at an equilibrium (where ``ω = 0``), so each has 2 degrees of
    freedom. In 3-D the normal direction is an uncontrollable mode with a zero
    eigenvalue, so the Riccati equation has no stabilising solution. ``T`` maps the 2-D
    tangent coordinates in the basis ``tangent_basis(q_eq)`` to the 3-D ones, and the S²
    rows (whose equations hold only on the tangent plane) are projected with ``Tᵀ``.
    A singular S² mass block (``Hat(q) Hat(q) m l²``) becomes invertible in this basis.

    Args:
        lin_module: the generated module of ``linearize`` (``M_lin``, ``A_lin``,
            ``B_lin``, ``LINEARIZATION``).
        x_eq: the equilibrium state (checked like ``simulate``'s ``x0``).
        u_eq: the equilibrium input.
        p: the parameters.
        Q: the weight of the reduced state: a scalar (times the identity), a vector (the
            diagonal) or a matrix.
        R: the weight of the input, in the same forms.
        a_eq: the accelerations at ``x_eq`` (zeros by default).
        tol: the tolerance of the ``x_eq`` check.
        module: the generated standard-form module of the same system
            (``generate_module`` of ``to_standard_form``, with the same layout: the same
            variables, velocities and inputs in the same order, checked). When
            given, ``(x_eq, u_eq)`` is checked to be an equilibrium, see below. When
            omitted, it is not checked and a ``UserWarning`` says so.
        eq_tol: the relative tolerance of the equilibrium check.

    The gain is only meaningful about an equilibrium: the linearization is evaluated at
    ``(x_eq, a_eq, u_eq)`` and the S² reduction assumes ``ω_eq = 0``. At a point that is
    not an equilibrium ``lqr`` still returns a gain with stable closed-loop eigenvalues,
    but the nonlinear loop settles somewhere else. With *module*, ``lqr`` therefore
    requires every velocity of ``x_eq`` to be zero (to *tol*) and the equation residual
    ``r = mass·a_eq + bias + G u_eq`` (S² rows projected on the tangent plane) to satisfy
    ``‖r‖∞ ≤ eq_tol · scale``, with ``scale`` the largest of ``‖bias‖∞``,
    ``‖G u_eq‖∞`` and ``‖A_lin T‖∞`` (the force per unit perturbation, which is nonzero
    at an equilibrium held by a potential even when the bias itself is zero there).
    Otherwise it raises ``ValueError`` with the residual and the accelerations
    ``accelerations(module, x_eq, u_eq, p)``.

    Raises:
        ValueError: bad arguments; a ``LINEARIZATION`` whose variables or inputs are not
            those of its ``LAYOUT`` in the same order; *module* with another layout than
            *lin_module* (the message names both variable and input orders); or
            ``(x_eq, u_eq)`` is not an equilibrium of *module*.
        np.linalg.LinAlgError: the reduced ``M_lin`` is singular.
    """
    try:
        from scipy.linalg import solve_continuous_are
    except ImportError as exc:  # pragma: no cover - depends on the environment
        raise ImportError("lqr needs scipy: pip install 'geomech[sim]'") from exc
    model = model_of(lin_module).require("M_lin", "A_lin", "B_lin")
    lin = getattr(lin_module, "LINEARIZATION", None)
    if not isinstance(lin, dict):
        raise TypeError("lqr: the module has no LINEARIZATION; generate it from linearize")
    x_eq = check_state(lin_module, x_eq, tol=tol, what="x_eq")
    u_eq = model.as_input(u_eq, "u_eq")
    a_eq = np.zeros(model.n_accel) if a_eq is None else np.asarray(a_eq, float).reshape(-1)
    if module is not None:
        _check_same_layout(model, model_of(module).require("mass", "bias", "input_matrix"))

    T, bases, labels = _reduction(model, lin, x_eq)
    M = lin_module.M_lin(x_eq, a_eq, u_eq, p)
    A = lin_module.A_lin(x_eq, a_eq, u_eq, p)
    B = lin_module.B_lin(x_eq, a_eq, u_eq, p)
    Mr = T.T @ M @ T
    try:
        Abar = np.linalg.solve(Mr, T.T @ A @ T)
        Bbar = np.linalg.solve(Mr, T.T @ B)
    except np.linalg.LinAlgError as exc:
        raise np.linalg.LinAlgError(
            f"lqr: the reduced M_lin is singular at x_eq (condition number"
            f" {np.linalg.cond(Mr):.3g})"
        ) from exc
    n = T.shape[1]
    Qm = _weight(Q, n, "Q")
    Rm = _weight(R, model.n_inputs, "R")
    if module is None:
        warnings.warn(
            "lqr: (x_eq, u_eq) is not checked to be an equilibrium; pass module=<the"
            " standard-form module> to check it (a gain about a non-equilibrium point"
            " stabilises the loop somewhere else)",
            UserWarning,
            stacklevel=2,
        )
    else:
        _check_equilibrium(model, module, x_eq, u_eq, a_eq, p, tol, eq_tol, A @ T)
    P = solve_continuous_are(Abar, Bbar, Qm, Rm)
    K = np.linalg.solve(Rm, Bbar.T @ P)
    return LQRController(K, P, Abar, Bbar, T, x_eq, u_eq, labels, model, bases)

    module: Any,
    x0: ArrayLike,
    t_span: float | tuple[float, float],
    p: dict,
    u: ArrayLike | Callable | None = None,
    method: str = "rkmk4",
    dt: float = 1e-3,
    *,
    rtol: float = 1e-10,
    atol: float = 1e-12,
    ivp_method: str = "DOP853",
    tol: float = STATE_TOL,
    output_every: int = 1,
) -> Trajectory:
    """Integrate a generated standard-form model from *x0* over *t_span*.

    Args:
        module: a module from ``geomech.codegen.generate_module`` of a
            ``to_standard_form`` result (``mass``, ``bias``, ``input_matrix``,
            ``kinematics``).
        x0: the initial state, flat in ``LAYOUT`` order. It must lie on the manifold
            (``|q| = 1``, ``q·ω = 0``, ``R`` orthogonal with ``det R = +1``, to *tol*),
            else ``ValueError`` naming the slot.
        t_span: ``(t0, tf)``, or ``tf`` for ``(0, tf)``.
        p: the parameters, keyed by printed symbol.
        u: ``None`` (zero inputs), a constant 1-D array, or a callable ``u(t, state)``
            returning the inputs (e.g. an ``LQRController``). The inputs must be finite;
            a callable that returns ``None`` raises ``TypeError``. A callable is
            evaluated once per RKMK4 stage, and ``traj.u`` records the values of each
            stored step's first stage (plus one call at the final state), so no state
            is evaluated twice. With ``"solve_ivp"`` it is evaluated at trial steps that
            may be rejected and at times that are not monotone, and once more at each
            stored (interpolated) state after the run: there it should be a pure
            function of ``(t, state)``. An exception raised by the callable propagates
            unchanged, as soon as it is raised, with a note giving ``t``.
        method: ``"rkmk4"`` (fixed step, default) or ``"solve_ivp"``.
        dt: the step of ``"rkmk4"`` (shortened so that it divides ``tf − t0``) and the
            output spacing of ``"solve_ivp"``.
        rtol: passed to ``scipy.integrate.solve_ivp``.
        atol: passed to ``scipy.integrate.solve_ivp``.
        ivp_method: the ``solve_ivp`` method (``"DOP853"``, ``"RK45"``, …).
        tol: the tolerance of the *x0* check.
        output_every: store every ``output_every``-th step of the ``dt`` grid (and
            always the last), to thin long runs; the integration itself is unchanged.

    The accelerations come from ``geomech.simulation.accelerations`` (the S² rows solved
    on the tangent plane). The stored states are on the manifold (``project_state``), so
    ``traj.state[-1]`` is a valid *x0* for the next segment. See ``docs/simulation.md``.
    """
    if method not in _METHODS:
        raise ValueError(f"simulate: method must be one of {_METHODS}, got {method!r}")
    model = model_of(module).require("mass", "bias", "input_matrix", "kinematics")
    x0 = check_state(module, x0, tol=tol, what="x0")
    t0, tf = _span(t_span)
    if not (dt > 0 and np.isfinite(dt)):
        raise ValueError(f"simulate: dt must be positive, got {dt}")
    if isinstance(output_every, bool) or not isinstance(output_every, (int, np.integer)):
        raise ValueError(f"simulate: output_every must be a positive int, got {output_every!r}")
    if output_every < 1:
        raise ValueError(f"simulate: output_every must be a positive int, got {output_every}")
    n_steps = max(1, int(np.ceil((tf - t0) / dt - 1e-9)))
    keep = np.unique(np.r_[np.arange(0, n_steps + 1, int(output_every)), n_steps])
    ufun = _input_function(model, u)
    p = dict(p or {})
    u0 = ufun(t0, x0)
    solve_accelerations(model, x0, u0, p, check=True)  # singular system: raise now
    if method == "rkmk4":
        t, states, inputs = _rkmk4(model, x0, t0, tf, n_steps, p, ufun, keep, u0)
        drift = constraint_drift(model, states)
        info = {"n_steps": n_steps, "dt": (tf - t0) / n_steps, "nfev": 4 * n_steps}
    else:
        t, raw, info = _solve_ivp(
            model, x0, t0, tf, n_steps, p, ufun, rtol, atol, ivp_method, keep
        )
        drift = constraint_drift(model, raw)
        states = np.array([project_state(model, x) for x in raw])
        # the stored states are interpolated, never evaluated during the run
        inputs = np.array([ufun(ti, x) for ti, x in zip(t, states)], dtype=float)
    info["output_every"] = int(output_every)
    inputs = inputs.reshape(len(t), model.n_inputs)
    return Trajectory(t, states, drift, method, p, module, info, inputs)

    def tangent_basis(q) -> np.ndarray:
    """An orthonormal basis ``[b₁ b₂]`` (3×2) of the plane orthogonal to the unit vector
    *q*, with ``b₁ × b₂ = q``. ``b₁`` is the coordinate axis least aligned with *q*,
    projected and normalised, so the basis is a deterministic function of *q*."""
    q = np.asarray(q, dtype=float).reshape(3)
    q = q / np.linalg.norm(q)
    e = np.eye(3)[int(np.argmin(np.abs(q)))]
    b1 = e - (e @ q) * q
    b1 /= np.linalg.norm(b1)
    b2 = np.cross(q, b1)
    return np.column_stack([b1, b2])
